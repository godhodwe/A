Instance.new("UICorner", autoSwingButton).CornerRadius = UDim.new(0, 8)
autoSwingButton.Visible = false  -- 처음엔 숨김

local autoSwingEnabled = false

submit.MouseButton1Click:Connect(function()
    if input.Text == "key1" then  -- 키 바꾸고 싶으면 여기 수정
        keyGui:Destroy()
        
        -- 로딩 화면 (기존 그대로)
        local Players = game:GetService("Players")
        local LocalPlayer = Players.LocalPlayer
        local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

        local ScreenGui = Instance.new("ScreenGui", PlayerGui)
        ScreenGui.Name = "SkibidiLoading"
        ScreenGui.ResetOnSpawn = false

        local Frame = Instance.new("Frame", ScreenGui)
        Frame.BackgroundColor3 = Color3.fromRGB(30,30,30)
        Frame.Size = UDim2.new(0, 420, 0, 180)
        Frame.Position = UDim2.new(0.5, -210, 0.5, -90)
        Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 12)

        local Title = Instance.new("TextLabel", Frame)
        Title.Text = "현존하는 최강 칼 스크립트"
        Title.Font = Enum.Font.GothamBlack
        Title.TextColor3 = Color3.fromRGB(255, 255, 255)
        Title.TextScaled = true
        Title.Size = UDim2.new(1, -20, 0, 60)
        Title.Position = UDim2.new(0, 10, 0, 10)
        Title.BackgroundTransparency = 1

        local LoadingText = Instance.new("TextLabel", Frame)
        LoadingText.Text = "제 스크립트를 써주셔서 감사합니다"
        LoadingText.Font = Enum.Font.Gotham
        LoadingText.TextColor3 = Color3.fromRGB(255, 255, 255)
        LoadingText.TextScaled = true
        LoadingText.Size = UDim2.new(1, -20, 0, 40)
        LoadingText.Position = UDim2.new(0, 10, 0, 80)
        LoadingText.BackgroundTransparency = 1

        local BarBack = Instance.new("Frame", Frame)
        BarBack.Size = UDim2.new(1, -40, 0, 20)
        BarBack.Position = UDim2.new(0, 20, 0.75, 0)
        BarBack.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        Instance.new("UICorner", BarBack).CornerRadius = UDim.new(0, 10)

        local BarFill = Instance.new("Frame", BarBack)
        BarFill.Size = UDim2.new(0, 0, 1, 0)
        BarFill.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
        Instance.new("UICorner", BarFill).CornerRadius = UDim.new(0, 10)

        for i = 1, 100 do
            BarFill.Size = UDim2.new(i/100, 0, 1, 0)
            wait(0.025)
        end

        for i = 0, 1, 0.05 do
            Frame.BackgroundTransparency = i
            Title.TextTransparency = i
            LoadingText.TextTransparency = i
            BarBack.BackgroundTransparency = i
            BarFill.BackgroundTransparency = i
            wait()
        end
        ScreenGui:Destroy()

        -- 메인 UI (KillAll + 자동 휘두르기 버튼 추가)
        local KillAllUI = Instance.new("ScreenGui", PlayerGui)
        KillAllUI.Name = "KillAllUI"
        KillAllUI.ResetOnSpawn = false

        local MainFrame = Instance.new("Frame", KillAllUI)
        MainFrame.Size = UDim2.new(0, 280, 0, 340)  -- 높이 더 늘림
        MainFrame.Position = UDim2.new(0.5, -140, 0.5, -170)
        MainFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
        MainFrame.Active = true
        MainFrame.Draggable = true
        Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 12)

        local Button = Instance.new("TextButton", MainFrame)
        Button.Size = UDim2.new(1, -20, 0, 50)
        Button.Position = UDim2.new(0, 10, 0, 10)
        Button.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        Button.TextColor3 = Color3.fromRGB(255, 0, 0)
        Button.Font = Enum.Font.GothamBold
        Button.TextScaled = true
        Button.Text = "누르고 시작"
        Instance.new("UICorner", Button).CornerRadius = UDim.new(0, 10)

        -- 자동 휘두르기 ON/OFF 버튼 추가
        local AutoSwingBtn = Instance.new("TextButton", MainFrame)
        AutoSwingBtn.Size = UDim2.new(1, -20, 0, 50)
        AutoSwingBtn.Position = UDim2.new(0, 10, 0, 70)
        AutoSwingBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
        AutoSwingBtn.Text = "자동 휘두르기 : OFF"
        AutoSwingBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
        AutoSwingBtn.Font = Enum.Font.GothamBold
        AutoSwingBtn.TextScaled = true
        Instance.new("UICorner", AutoSwingBtn).CornerRadius = UDim.new(0, 10)

        local Credit = Instance.new("TextLabel", MainFrame)
        Credit.Size = UDim2.new(1, -20, 0, 20)
        Credit.Position = UDim2.new(0, 10, 1, -30)
        Credit.BackgroundTransparency = 1
        Credit.Text = "리휸춘이가 만들었습니다 ㅅㅂㅋㅋ"
        Credit.TextColor3 = Color3.fromRGB(150, 150, 150)
        Credit.TextScaled = true
        Credit.Font = Enum.Font.Gotham
        Credit.TextXAlignment = Enum.TextXAlignment.Center

        -- 리치 조절
        local ReachLabel = Instance.new("TextLabel", MainFrame)
        ReachLabel.Size = UDim2.new(1, -20, 0, 20)
        ReachLabel.Position = UDim2.new(0, 10, 0, 130)
        ReachLabel.BackgroundTransparency = 1
        ReachLabel.TextColor3 = Color3.new(1,1,1)
        ReachLabel.TextScaled = true
        ReachLabel.Text = "리치: 50"

        local ReachInput = Instance.new("TextBox", MainFrame)
        ReachInput.Size = UDim2.new(1, -20, 0, 30)
        ReachInput.Position = UDim2.new(0, 10, 0, 155)
        ReachInput.BackgroundColor3 = Color3.fromRGB(50,50,50)
        ReachInput.TextColor3 = Color3.new(1,1,1)
        ReachInput.Text = "50"
        ReachInput.TextScaled = true
        Instance.new("UICorner", ReachInput).CornerRadius = UDim.new(0, 8)

        -- 특정 타겟
        local TargetLabel = Instance.new("TextLabel", MainFrame)
        TargetLabel.Size = UDim2.new(1, -20, 0, 20)
        TargetLabel.Position = UDim2.new(0, 10, 0, 190)
        TargetLabel.BackgroundTransparency = 1
        TargetLabel.TextColor3 = Color3.new(1,1,1)
        TargetLabel.TextScaled = true
        TargetLabel.Text = "특정 타겟 (부분 입력 OK)"

        local TargetInput = Instance.new("TextBox", MainFrame)
        TargetInput.Size = UDim2.new(1, -20, 0, 30)
        TargetInput.Position = UDim2.new(0, 10, 0, 215)
        TargetInput.BackgroundColor3 = Color3.fromRGB(50,50,50)
        TargetInput.TextColor3 = Color3.new(1,1,1)
        TargetInput.PlaceholderText = "닉네임 (없으면 모두)"
        TargetInput.TextScaled = true
        Instance.new("UICorner", TargetInput).CornerRadius = UDim.new(0, 8)

        -- 친추 보호
        local FriendSafeLabel = Instance.new("TextLabel", MainFrame)
        FriendSafeLabel.Size = UDim2.new(1, -20, 0, 20)
        FriendSafeLabel.Position = UDim2.new(0, 10, 0, 250)
        FriendSafeLabel.BackgroundTransparency = 1
        FriendSafeLabel.TextColor3 = Color3.new(1,1,1)
        FriendSafeLabel.TextScaled = true
        FriendSafeLabel.Text = "친추 보호: ON"

        local FriendSafeButton = Instance.new("TextButton", MainFrame)
        FriendSafeButton.Size = UDim2.new(1, -20, 0, 30)
        FriendSafeButton.Position = UDim2.new(0, 10, 0, 275)
        FriendSafeButton.BackgroundColor3 = Color3.fromRGB(50,50,50)
        FriendSafeButton.Text = "ON"
        FriendSafeButton.TextColor3 = Color3.new(1,1,1)
        FriendSafeButton.TextScaled = true
        Instance.new("UICorner", FriendSafeButton).CornerRadius = UDim.new(0, 8)

        local Enabled = false
        local Reach = 50
        local TargetPlayer = nil
        local FriendSafe = true

        Button.MouseButton1Click:Connect(function()
            Enabled = not Enabled
            Button.Text = Enabled and "킬 : 켜짐" or "킬 : 꺼짐"
            Button.TextColor3 = Enabled and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
        end)

        -- 자동 휘두르기 버튼
        AutoSwingBtn.MouseButton1Click:Connect(function()
            autoSwingEnabled = not autoSwingEnabled
            AutoSwingBtn.Text = "자동 휘두르기 : " .. (autoSwingEnabled and "ON" or "OFF")
            AutoSwingBtn.BackgroundColor3 = autoSwingEnabled and Color3.fromRGB(0, 170, 80) or Color3.fromRGB(60, 60, 60)
        end)

        ReachInput.FocusLost:Connect(function()
            local num = tonumber(ReachInput.Text)
            if num and num > 0 then
                Reach = num
                ReachLabel.Text = "리치: " .. num
            end
        end)

        TargetInput.FocusLost:Connect(function()
            TargetPlayer = TargetInput.Text ~= "" and TargetInput.Text:lower() or nil
        end)

        FriendSafeButton.MouseButton1Click:Connect(function()
            FriendSafe = not FriendSafe
            FriendSafeButton.Text = FriendSafe and "ON" or "OFF"
            FriendSafeButton.BackgroundColor3 = FriendSafe and Color3.fromRGB(50,50,50) or Color3.fromRGB(80,80,80)
            FriendSafeLabel.Text = "친추 보호: " .. (FriendSafe and "ON" or "OFF")
        end)

        -- 자동 휘두르기 로직 추가
        game:GetService("RunService").Heartbeat:Connect(function()
            if not autoSwingEnabled then return end

            local char = LocalPlayer.Character
            if not char then return end

            local tool = char:FindFirstChildOfClass("Tool")
            if not tool then return end

            -- 무조건 휘두르기 (거리 상관없이)
            pcall(function()
                tool:Activate()
            end)
        end)

        -- 기존 킬오올 로직 (RenderStepped)
        game:GetService("RunService").RenderStepped:Connect(function()
            if not Enabled then return end
            for _, plr in pairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and (not TargetPlayer or plr.Name:lower():find(TargetPlayer)) then
                    if FriendSafe and plr:IsFriendsWith(LocalPlayer.UserId) then
                        continue
                    end

                    local char = plr.Character
                    if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0 then
                        if LocalPlayer:DistanceFromCharacter(char.HumanoidRootPart.Position) <= Reach then
                            local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
                            if tool and tool:FindFirstChild("Handle") then
                                tool:Activate()
                                for _, part in pairs(char:GetChildren()) do
                                    if part:IsA("BasePart") then
                                        firetouchinterest(tool.Handle, part, 0)
                                        firetouchinterest(tool.Handle, part, 1)
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end)
    else
        game.StarterGui:SetCore("SendNotification", {
            Title = "잘못된 키",
            Text = "틀렸어!",
            Duration = 5
        })
    end
end)
